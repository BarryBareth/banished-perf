static TerrainPoint_t o_TerrainPoint;
// Pathfinder message dispatcher exe+0xab900 (message 5 = solve into a result slot) and result-slot allocation
// exe+0xab710 (grows the slot array, copying every slot's path): path memory, see t_pathMem.
using PathMessage_t = void(__fastcall*)(void* pf, const int* msg);
static PathMessage_t o_PathMessage;
static void __fastcall h_PathMessage(void* pf, const int* msg) {
    if (msg[0] != 5) return o_PathMessage(pf, msg);
    t_pathMem++;
    o_PathMessage(pf, msg);
    t_pathMem--;
}
using PathSlotAlloc_t = int*(__fastcall*)(void* pf, int* out);
static PathSlotAlloc_t o_PathSlotAlloc;
static int* __fastcall h_PathSlotAlloc(void* pf, int* out) {
    t_pathMem++;
    int* r = o_PathSlotAlloc(pf, out);
    t_pathMem--;
    return r;
}
// For bperf (dbghelp.dll): it calls the game's pool / vector code for path results itself, inside this scope.
extern "C" void __stdcall BmpPathMemory(int enter) { t_pathMem += enter ? 1 : -1; }
